    21→    21→/**
    22→    22→ * powerbi.visuals.ValidatorType is a `const enum` (Min=0, Max=1, Required=2).
    23→    23→ * The build chain inlines it, which would break under isolatedModules if we
    24→    24→ * referenced the enum symbol. We use numeric literals instead.
    25→    25→ */
    26→    26→const VT_MIN = 0;
    27→    27→const VT_MAX = 1;
    28→    28→
    29→    29→/**
    30→    30→ * Build an IEnumMember-shaped value object for ItemDropdown / AutoDropdown.
    31→    31→ * displayName is a plain string (DisplayText form of DisplayNameGetter).
    32→    32→ */
    33→    33→function enumItem(value: string, displayName: string): powerbi.IEnumMember {
    34→    34→    return { value, displayName } as powerbi.IEnumMember;
    35→    35→}
    36→    36→
    37→    37→/**
    38→    38→ * Read a string out of an ItemDropdown's `value` (IEnumMember). Falls back to
    39→    39→ * the provided default when the slice isn't populated yet.
    40→    40→ */
    41→    41→function readEnumValue(
    42→    42→    slice: formattingSettings.ItemDropdown,
    43→    43→    defaultValue: string
    44→    44→): string {
    45→    45→    const v = slice && slice.value;
    46→    46→    if (v && typeof v === "object" && typeof (v as any).value === "string") {
    47→    47→        return (v as any).value as string;
    48→    48→    }
    49→    49→    if (typeof v === "string") {
    50→    50→        return v;
    51→    51→    }
    52→    52→    return defaultValue;
    53→    53→}
    54→    54→
    55→    55→/**
    56→    56→ * Read a boolean slice value (ToggleSwitch). Falls back to default when the
    57→    57→ * slice isn't populated yet (Power BI passes undefined for "default" states).
    58→    58→ */
    59→    59→function readBool(
    60→    60→    slice: formattingSettings.ToggleSwitch | undefined,
    61→    61→    defaultValue: boolean
    62→    62→): boolean {
    63→    63→    if (slice && typeof slice.value === "boolean") {
    64→    64→        return slice.value;
    65→    65→    }
    66→    66→    return defaultValue;
    67→    67→}
    68→    68→
    69→    69→function readNumber(
    70→    70→    slice: formattingSettings.NumUpDown | undefined,
    71→    71→    defaultValue: number
    72→    72→): number {
    73→    73→    if (slice && typeof slice.value === "number" && isFinite(slice.value)) {
    74→    74→        return slice.value;
    75→    75→    }
    76→    76→    return defaultValue;
    77→    77→}
    78→    78→
    79→    79→function readString(
    80→    80→    slice:
    81→    81→        | formattingSettings.TextInput
    82→    82→        | formattingSettings.FontPicker
    83→    83→        | undefined,
    84→    84→    defaultValue: string
    85→    85→): string {
    86→    86→    if (slice && typeof slice.value === "string" && slice.value.length > 0) {
    87→    87→        return slice.value;
    88→    88→    }
    89→    89→    return defaultValue;
    90→    90→}
    91→    91→
    92→    92→function readColor(
    93→    93→    slice: formattingSettings.ColorPicker | undefined,
    94→    94→    defaultValue: string
    95→    95→): string {
    96→    96→    if (slice && slice.value && typeof slice.value.value === "string") {
    97→    97→        return slice.value.value;
    98→    98→    }
    99→    99→    return defaultValue;
   100→   100→}
