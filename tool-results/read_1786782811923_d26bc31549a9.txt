    18→/** Monotonic counter for unique clip-path IDs (NOT Math.random — predictable). */
    19→let clipIdCounter = 0;
    20→function nextClipId(prefix: string): string {
    21→    clipIdCounter += 1;
    22→    return `${prefix}-${clipIdCounter}`;
    23→}

    25→/** Shared canvas for measureText — created lazily, reused. */
    26→let measureCanvas: HTMLCanvasElement | null = null;
    27→function getMeasureCanvas(): HTMLCanvasElement {
    28→    if (!measureCanvas) {
    29→        measureCanvas = document.createElement("canvas");
    30→        const ctx = measureCanvas.getContext("2d");
    31→        if (ctx) {
    32→            ctx.font = "11px 'Segoe UI', sans-serif";
    33→        }
    34→    }
    35→    return measureCanvas;
    36→}

    38→/** Measure pixel width of a text string with given font. */
    39→function measureTextWidth(
    40→    text: string,
    41→    fontFamily: string,
    42→    fontSize: number,
    43→    bold: boolean
    44→): number {
    45→    const canvas = getMeasureCanvas();
    46→    const ctx = canvas.getContext("2d");
    47→    if (!ctx) return text.length * fontSize * 0.55;
    48→    ctx.font = `${bold ? "bold " : ""}${fontSize}px '${fontFamily}', sans-serif`;
    49→    return ctx.measureText(text).width;
    50→}

    52→/** Convert "#RRGGBB" to its rgb triplet for interpolation. */
    53→function hexToRgb(hex: string): [number, number, number] {
    54→    let h = hex.replace("#", "").trim();
    55→    if (h.length === 3) {
    56→        h = h
    57→            .split("")
    58→            .map((c) => c + c)
    59→            .join("");
    60→    }
    61→    const n = parseInt(h, 16);
    62→    if (isNaN(n)) return [0, 0, 0];
    63→    return [(n >> 16) & 0xff, (n >> 8) & 0xff, n & 0xff];
    64→}
    65→
    66→function rgbToHex(r: number, g: number, b: number): string {
    67→    const h = (v: number) =>
    68→        Math.max(0, Math.min(255, Math.round(v)))
    69→            .toString(16)
    70→            .padStart(2, "0");
    71→    return `#${h(r)}${h(g)}${h(b)}`;
    72→}

    74→/**
    75→ * Multi-stop gradient interpolation: given N colors, return the interpolated
    76→ * color at position `t` (0..1). Used to color bars by Pareto rank.
    77→ */
    78→function interpolateGradient(
    79→    colors: string[],
    80→    t: number
    81→): string {
    82→    if (colors.length === 0) return "#1F77B4";
    83→    if (colors.length === 1) return colors[0];
    84→    const clamped = Math.max(0, Math.min(1, t));
    85→    const segments = colors.length - 1;
    86→    const scaled = clamped * segments;
    87→    const idx = Math.min(segments - 1, Math.floor(scaled));
    88→    const localT = scaled - idx;
    89→    const c1 = hexToRgb(colors[idx]);
    90→    const c2 = hexToRgb(colors[idx + 1]);
    91→    return rgbToHex(
    92→        c1[0] + (c2[0] - c1[0]) * localT,
    93→        c1[1] + (c2[1] - c1[1]) * localT,
    94→        c1[2] + (c2[2] - c1[2]) * localT
    95→    );
    96→}

    98→/**
    99→ * FIX #1: Selected bar keeps its original color.
   100→ *
   101→ * Resolve the fill color of a bar EXACTLY as if it were not selected.
   102→ * - Default mode: settings.bars.defaultColor
   103→ * - Single mode: settings.bars.singleColor
   104→ * - Gradient mode: interpolated across categories by rank
   105→ * - Category mode: leaves it up to the caller (we fall back to default)
   106→ */
   107→function resolveBarColor(
   108→    point: ParetoDataPoint,
   109→    settings: ResolvedSettings,
   110→    totalPoints: number
   111→): string {
   112→    const mode = settings.bars.colorMode;
   113→    switch (mode) {
   114→        case "Single":
   115→            return settings.bars.singleColor;
   116→        case "Gradient": {
   117→            const colors = settings.bars.gradientColors;
   118→            if (colors.length === 0) return settings.bars.defaultColor;
   119→            // Position by rank: t=0 for the first bar, t=1 for the last.
   120→            const t =
   121→                totalPoints > 1 ? (point.rank - 1) / (totalPoints - 1) : 0;
   122→            return interpolateGradient(colors, t);
   123→        }
   124→        case "Category":
   125→            // Category-based coloring isn't persisted across renders here;
   126→            // we fall back to the default color (consistent v3.0 behavior).
   127→            return settings.bars.defaultColor;
   128→        case "Default":
   129→        default:
   130→            return settings.bars.defaultColor;
   131→    }
   132→}

   134→/**
   135→ * FIX #1: Selected bar opacity.
   136→ *
   137→ * - Selected: base opacity (settings.bars.opacity / 100). NOT 1.0 — the bar
   138→ *   looks exactly like an unselected bar.
   139→ * - Dimmed (unselected in a selection state): fixed 0.3.
   140→ * - Highlighted (cross-filter): if hasHighlight && !isHighlighted → 0.3.
   141→ * - Otherwise: base opacity.
   142→ */
   143→function resolveBarOpacity(
   144→    point: ParetoDataPoint,
   145→    settings: ResolvedSettings,
   146→    hasHighlight: boolean
   147→): number {
   148→    const base = settings.bars.opacity / 100;
   149→    if (point.isSelected) return base;
   150→    if (point.isDimmed) return 0.3;
   151→    if (hasHighlight && !point.isHighlighted) return 0.3;
   152→    return base;
   153→}

   155→/** Build the dash-array for a given line style + width. */
   156→function dashArray(style: "Solid" | "Dashed" | "Dotted", width: number): string {
   157→    switch (style) {
   158→        case "Dashed":
   159→            return `${width * 3},${width * 2}`;
   160→        case "Dotted":
   161→            return `${width},${width}`;
   162→        case "Solid":
   163→        default:
   164→            return "none";
   165→    }
   166→}

   168→/** Word-wrap a label into at most `maxLines` lines that each fit maxWidth. */
   169→function wrapLabel(
   170→    text: string,
   171→    maxWidth: number,
   172→    maxLines: number,
   173→    fontFamily: string,
   174→    fontSize: number,
   175→    bold: boolean
   176→): string[] {
   177→    if (!text) return [];
   178→    if (maxWidth <= 0) return [text];
   179→    const words = String(text).split(/\s+/).filter((w) => w.length > 0);
   180→    if (words.length === 0) return [];
   181→    const lines: string[] = [];
   182→    let current = words[0];
   183→    for (let i = 1; i < words.length; i++) {
   184→        const candidate = `${current} ${words[i]}`;
   185→        const w = measureTextWidth(candidate, fontFamily, fontSize, bold);
   186→        if (w <= maxWidth) {
   187→            current = candidate;
   188→        } else {
   189→            lines.push(current);
   190→            current = words[i];
   191→            if (lines.length >= maxLines - 1) {
   192→                // Last line: append the rest, truncated with ellipsis if needed.
   193→                const rest = [current].concat(words.slice(i + 1)).join(" ");
   194→                let final = rest;
   195→                while (
   196→                    final.length > 1 &&
   197→                    measureTextWidth(final + "…", fontFamily, fontSize, bold) >
   198→                        maxWidth
   199→                ) {
   200→                    final = final.slice(0, -1);
   201→                }
   202→                if (final.length < rest.length) final = final + "…";
   203→                lines.push(final);
   204→                return lines.slice(0, maxLines);
   205→            }
   206→        }
   207→    }
   208→    lines.push(current);
   209→    return lines.slice(0, maxLines);
   210→}

   226→/**
   227→ * Compute the effective max lines for word-wrap given the mode.
   228→ * "Auto" → effectively unlimited (passes 100), bottom padding uses the actual
   229→ *          max lines after wrapping.
   230→ * "Number" → use the configured maxLines value.
   231→ */
   232→function effectiveMaxLines(settings: ResolvedSettings): number {
   233→    if (settings.xAxis.maxLinesMode === "Auto") return 100;
   234→    return Math.max(1, Math.min(10, settings.xAxis.maxLines));
   235→}

   263→/** Build a class string for a bar based on its selection state. */
   264→function barClass(point: ParetoDataPoint): string {
   265→    const parts = ["bar"];
   266→    if (point.isSelected) parts.push("selected");
   267→    else if (point.isDimmed) parts.push("dimmed");
   268→    return parts.join(" ");
   269→}

   271→/**
   272→ * Main entry point. Renders the full Pareto chart into the provided SVG.
   273→ * Wipes all existing children first.
   274→ */
   275→export function renderPareto(opts: RenderOptions): void {
   276→    const { svg, model, settings, viewport, isRtl } = opts;
   277→    while (svg.firstChild) svg.removeChild(svg.firstChild);
   278→
   279→    const svgSel = d3.select(svg) as SVGSel;
   280→    svgSel
   281→        .attr("class", "industrial-pareto")
   282→        .attr("width", viewport.width)
   283→        .attr("height", viewport.height)
   284→        .attr("viewBox", `0 0 ${viewport.width} ${viewport.height}`)
   285→        .attr("dir", isRtl ? "rtl" : "ltr");
   286→
   287→    // Background
   288→    if (settings.background.show) {
   289→        const alpha = 1 - settings.background.transparency / 100;
   290→        svgSel
   291→            .append("rect")
   292→            .attr("x", 0)
   293→            .attr("y", 0)
   294→            .attr("width", viewport.width)
   295→            .attr("height", viewport.height)
   296→            .attr("fill", settings.background.color)
   297→            .attr("fill-opacity", alpha);
   298→    }
   299→
   300→    if (!model.hasData || model.points.length === 0) {
   301→        svgSel
   302→            .append("text")
   303→            .attr("x", viewport.width / 2)
   304→            .attr("y", viewport.height / 2)
   305→            .attr("text-anchor", "middle")
   306→            .attr("fill", "#888")
   307→            .style("font-family", "Segoe UI, sans-serif")
   308→            .style("font-size", "14px")
   309→            .text("No data to display");
   310→        return;
   311→    }
   312→
   313→    const points = model.points;
   314→    const totalPoints = points.length;
   315→
   316→    /* ------------------------- Layout / padding ------------------------- */
   317→    // Default paddings. We then EXPAND them based on axis visibility, label
   318→    // heights, and rotated/wrapped X labels.
   319→    const padTop = 16;
   320→    const padRight = settings.secondaryAxis.show ? 56 : 24;
   321→    const padLeftBase = settings.yAxis.show ? 60 : 16;
   322→
   323→    // Measure Y-axis tick widths to size the left padding correctly.
   324→    const yMax = model.maxValue;
   325→    const yTickValues = computeNiceTicks(0, yMax, 5);
   326→    let yTickWidth = 0;
   327→    if (settings.yAxis.show) {
   328→        for (const v of yTickValues) {
   329→            const lbl = formatAxisValue(
   330→                v,
   331→                settings.yAxis.valueFormat,
   332→                model.looksLikeDuration
   333→            );
   334→            const w = measureTextWidth(
   335→                lbl,
   336→                settings.yAxis.fontFamily,
   337→                settings.yAxis.fontSize,
   338→                settings.yAxis.bold
   339→            );
   340→            if (w > yTickWidth) yTickWidth = w;
   341→        }
   342→    }
   343→
   344→    const padLeft = padLeftBase + yTickWidth;
   345→
   346→    // X-axis label height calculation depends on rotation + word-wrap mode.
   347→    // We need a working inner width to wrap against, so compute it now.
   348→    const innerWidth = Math.max(
   349→        50,
   350→        viewport.width - padLeft - padRight
   351→    );
   352→
   353→    // Wrap X labels to determine the bottom padding.
   354→    let maxLabelLines = 1;
   355→    let xLabelHeight = settings.xAxis.fontSize + 8;
   356→    if (settings.xAxis.show) {
   357→        const rotation = settings.xAxis.textRotation;
   358→        const fontFamily = settings.xAxis.fontFamily;
   359→        const fontSize = settings.xAxis.fontSize;
   360→        const bold = settings.xAxis.bold;
   361→        if (rotation === "Rotation0" && settings.xAxis.wordWrap) {
   362→            const maxLinesCfg = effectiveMaxLines(settings);
   363→            const maxLabelWidth = Math.max(
   364→                20,
   365→                innerWidth / totalPoints - 4
   366→            );
   367→            for (const p of points) {
   368→                const lines = wrapLabel(
   369→                    p.category,
   370→                    maxLabelWidth,
   371→                    maxLinesCfg,
   372→                    fontFamily,
   373→                    fontSize,
   374→                    bold
   375→                );
   376→                if (lines.length > maxLabelLines) maxLabelLines = lines.length;
   377→            }
   378→            xLabelHeight = maxLabelLines * (fontSize + 2) + 8;
   379→        } else if (rotation === "Rotation0") {
   380→            // Word wrap off — labels may be thinned (existing v3.0 behavior).
   381→            xLabelHeight = fontSize + 8;
   382→        } else if (rotation === "Rotation45") {
   383→            // Estimate height from longest label length (rotated 45°).
   384→            let longest = 0;
   385→            for (const p of points) {
   386→                const w = measureTextWidth(
   387→                    p.category,
   388→                    fontFamily,
   389→                    fontSize,
   390→                    bold
   391→                );
   392→                if (w > longest) longest = w;
   393→            }
   394→            xLabelHeight = longest * 0.707 + 8;
   395→        } else {
   396→            // Rotation90 — full height of the longest label.
   397→            let longest = 0;
   398→            for (const p of points) {
   399→                const w = measureTextWidth(
   400→                    p.category,
   401→                    fontFamily,
   402→                    fontSize,
   403→                    bold
   404→                );
   405→                if (w > longest) longest = w;
   406→            }
   407→            xLabelHeight = longest + 8;
   408→        }
   409→    }
   410→
   411→    const padBottom = (settings.xAxis.show ? xLabelHeight : 16) + 12;
   412→    const innerHeight = Math.max(
   413→        40,
   414→        viewport.height - padTop - padBottom
   415→    );
   416→
   417→    /* ----------------------------- Clip paths --------------------------- */
   418→    const plotClipId = nextClipId("plotClip");
   419→    const defs = svgSel.append("defs");
   420→    defs.append("clipPath")
   421→        .attr("id", plotClipId)
   422→        .append("rect")
   423→        .attr("x", 0)
   424→        .attr("y", 0)
   425→        .attr("width", innerWidth)
   426→        .attr("height", innerHeight);
   427→
   428→    /* --------------------------- Root groups ---------------------------- */
   429→    // Plot group is clipped (so bars/curve don't spill into axes).
   430→    // Axes group is NOT clipped (so labels can extend slightly).
   431→    const plotGroup = svgSel
   432→        .append("g")
   433→        .attr(
   434→            "transform",
   435→            `translate(${padLeft},${padTop})`
   436→        )
   437→        .attr("clip-path", `url(#${plotClipId})`);
   438→
   439→    const axesGroup = svgSel
   440→        .append("g")
   441→        .attr("transform", `translate(${padLeft},${padTop})`);
   442→
   443→    /* ------------------------------ Scales ------------------------------ */
   444→    // Bars on X use ScaleBand with configurable gap.
   445→    const gapFraction = Math.max(0, Math.min(0.9, settings.bars.barGap / 100));
   446→    const xScale = d3
   447→        .scaleBand<string>()
   448→        .domain(points.map((p) => p.category))
   449→        .range(isRtl ? [innerWidth, 0] : [0, innerWidth])
   450→        .padding(gapFraction);
   451→
   452→    // Y axis: linear, value domain.
   453→    const yScale = d3
   454→        .scaleLinear<number, number>()
   455→        .domain([0, yMax * 1.05 || 1])
   456→        .range([innerHeight, 0]);
   457→
   458→    // Y2 axis: cumulative percentage (0..100, or user-configured min/max).
   459→    const y2Min = settings.secondaryAxis.min;
   460→    const y2Max = settings.secondaryAxis.max;
   461→    const y2Scale = d3
   462→        .scaleLinear<number, number>()
   463→        .domain([y2Min, y2Max])
   464→        .range([innerHeight, 0]);
   465→
   466→    /* ------------------------------ Grid lines -------------------------- */
   467→    const gridGroup = plotGroup.append("g").attr("class", "grid");
   468→    gridGroup
   469→        .selectAll("line.grid-y")
   470→        .data(yTickValues)
   471→        .enter()
   472→        .append("line")
   473→        .attr("class", "grid-y")
   474→        .attr("x1", 0)
   475→        .attr("x2", innerWidth)
   476→        .attr("y1", (d) => yScale(d))
   477→        .attr("y2", (d) => yScale(d));
   478→
   479→    /* ------------------------------ Bars -------------------------------- */
   480→    const barsGroup = plotGroup.append("g").attr("class", "bars");
   481→    const barSel = barsGroup
   482→        .selectAll<SVGRectElement, ParetoDataPoint>("rect.bar")
   483→        .data(points)
   484→        .enter()
   485→        .append("rect")
   486→        .attr("class", (d) => barClass(d))
   487→        .attr(
   488→            "x",
   489→            (d) => (xScale(d.category) as number) + xScale.bandwidth() * 0
   490→        )
   491→        .attr("y", (d) => {
   492→            // Use highlightedValue when present, otherwise rawValue.
   493→            const v = d.isHighlighted && d.highlightedValue != null
   494→                ? d.highlightedValue
   495→                : d.rawValue;
   496→            return yScale(Math.max(0, v));
   497→        })
   498→        .attr("width", xScale.bandwidth())
   499→        .attr("height", (d) => {
   500→            const v = d.isHighlighted && d.highlightedValue != null
   501→                ? d.highlightedValue
   502→                : d.rawValue;
   503→            const y = yScale(Math.max(0, v));
   504→            return Math.max(0, innerHeight - y);
   505→        })
   506→        .attr("fill", (d) => resolveBarColor(d, settings, totalPoints))
   507→        .attr("fill-opacity", (d) =>
   508→            resolveBarOpacity(d, settings, model.hasHighlight)
   509→        )
   510→        .style("cursor", "pointer");
   511→
   512→    // Wire bar events. The visual owns the handlers so it can dispatch
   513→    // to its selection manager / tooltip service.
   514→    barSel.each(function (point) {
   515→        const barEl = this as SVGRectElement;
   516→        barEl.addEventListener("click", (ev) => {
   517→            opts.onBarClick(point, ev as MouseEvent, barEl);
   518→        });
   519→        barEl.addEventListener("contextmenu", (ev) => {
   520→            opts.onBarContextMenu(point, ev as MouseEvent, barEl);
   521→        });
   522→        barEl.addEventListener("mousemove", (ev) => {
   523→            opts.onBarMouseMove(point, ev as MouseEvent, barEl);
   524→        });
   525→        barEl.addEventListener("mouseleave", () => {
   526→            opts.onBarMouseLeave();
   527→        });
   528→    });
   529→
   530→    /* -------------------------- Bar data labels ------------------------- */
   531→    if (settings.barDataLabels.show) {
   532→        const dl = settings.barDataLabels;
   533→        const labelGroup = plotGroup
   534→            .append("g")
   535→            .attr("class", "bar-labels")
   536→            .style("pointer-events", "none");
   537→        const labelSel = labelGroup
   538→            .selectAll<SVGTextElement, ParetoDataPoint>("text.bar-label")
   539→            .data(points)
   540→            .enter()
   541→            .append("text")
   542→            .attr("class", "bar-label")
   543→            .attr("font-family", dl.fontFamily)
   544→            .attr("font-size", dl.fontSize)
   545→            .attr("font-weight", dl.bold ? "bold" : "normal")
   546→            .attr("font-style", dl.italic ? "italic" : "normal")
   547→            .attr("text-decoration", dl.underline ? "underline" : "none")
   548→            .attr("fill", dl.color);
   549→
   550→        labelSel.each(function (point) {
   551→            const txt = this as SVGTextElement;
   552→            const v =
   553→                point.isHighlighted && point.highlightedValue != null
   554→                    ? point.highlightedValue
   555→                    : point.rawValue;
   556→            const text = formatAxisValue(v, dl.format, model.looksLikeDuration);
   557→            const bandX = xScale(point.category) as number;
   558→            const bandW = xScale.bandwidth();
   559→            const y0 = yScale(Math.max(0, v));
   560→            let labelX: number;
   561→            let labelY: number;
   562→            let anchor: "start" | "middle" | "end";
   563→            switch (dl.position) {
   564→                case "Above":
   565→                    labelX = bandX + bandW / 2;
   566→                    labelY = y0 - 6;
   567→                    anchor = "middle";
   568→                    break;
   569→                case "Below":
   570→                    labelX = bandX + bandW / 2;
   571→                    labelY = y0 + dl.fontSize + 2;
   572→                    anchor = "middle";
   573→                    break;
   574→                case "InsideEnd":
   575→                    labelX = bandX + bandW / 2;
   576→                    labelY = y0 + dl.fontSize;
   577→                    anchor = "middle";
   578→                    break;
   579→                case "InsideBase":
   580→                    labelX = bandX + bandW / 2;
   581→                    labelY = innerHeight - 4;
   582→                    anchor = "middle";
   583→                    break;
   584→                default:
   585→                    labelX = bandX + bandW / 2;
   586→                    labelY = y0 - 6;
   587→                    anchor = "middle";
   588→            }
   589→            // Background rect (if enabled). Rendered as a sibling rect BEFORE
   590→            // the text element so it sits underneath.
   591→            if (dl.showBackground) {
   592→                const w = measureTextWidth(

   998→/** Draw a marker (circle/square/triangle/diamond) inside a <g>. */
   999→function drawMarker(
  1000→    parent: GroupSel,
  1001→    shape: string,
  1002→    size: number,
  1003→    color: string
  1004→): void {
  1005→    const r = size / 2;
  1006→    switch (shape) {
  1007→        case "Square":
  1008→            parent
  1009→                .append("rect")
  1010→                .attr("x", -r)
  1011→                .attr("y", -r)
  1012→                .attr("width", size)
  1013→                .attr("height", size)
  1014→                .attr("fill", color);
  1015→            break;
  1016→        case "Triangle":
  1017→            parent
  1018→                .append("polygon")
  1019→                .attr(
  1020→                    "points",
  1021→                    `0,${-r} ${r},${r} ${-r},${r}`
  1022→                )
  1023→                .attr("fill", color);
  1024→            break;
  1025→        case "Diamond":
  1026→            parent
  1027→                .append("polygon")
  1028→                .attr(
  1029→                    "points",
  1030→                    `0,${-r} ${r},0 0,${r} ${-r},0`
  1031→                )
  1032→                .attr("fill", color);
  1033→            break;
  1034→        case "Circle":
  1035→        default:
  1036→            parent
  1037→                .append("circle")
  1038→                .attr("r", r)
  1039→                .attr("fill", color);
  1040→    }
  1041→}

  1043→/** Compute "nice" tick values for a [min, max] range. Pure numeric, no D3 axis. */
  1044→function computeNiceTicks(min: number, max: number, count: number): number[] {
  1045→    if (min === max) {
  1046→        return [min];
  1047→    }
  1048→    const span = max - min;
  1049→    const step0 = span / count;
  1050→    const mag = Math.pow(10, Math.floor(Math.log10(step0)));
  1051→    const norm = step0 / mag;
  1052→    let step: number;
  1053→    if (norm < 1.5) step = 1 * mag;
  1054→    else if (norm < 3) step = 2 * mag;
  1055→    else if (norm < 7) step = 5 * mag;
  1056→    else step = 10 * mag;
  1057→    const start = Math.ceil(min / step) * step;
  1058→    const ticks: number[] = [];
  1059→    for (let v = start; v <= max + step * 0.5; v += step) {
  1060→        ticks.push(Math.round(v * 1e9) / 1e9);
  1061→    }
  1062→    return ticks;
  1063→}

  1065→/** Compute the visible-label step for labelDensity thinning. */
  1066→function computeLabelStep(
  1067→    total: number,
  1068→    density: "ShowAll" | "Auto" | "Sparse"
  1069→): number {
  1070→    if (density === "ShowAll") return 1;
  1071→    if (total <= 8) return 1;
  1072→    if (density === "Auto") {
  1073→        // ~10 labels max
  1074→        return Math.max(1, Math.ceil(total / 10));
  1075→    }
  1076→    // Sparse: ~5 labels max
  1077→    return Math.max(1, Math.ceil(total / 5));
  1078→}
