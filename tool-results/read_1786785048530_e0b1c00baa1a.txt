    20→    20→/**
    21→    21→ * powerbi.visuals.ValidatorType is a `const enum` (Min=0, Max=1, Required=2).
    22→    22→ * The build chain inlines it, which would break under isolatedModules if we
    23→    23→ * referenced the enum symbol. We use numeric literals instead.
    24→    24→ */
    25→    25→const VT_MIN = 0;
    26→    26→const VT_MAX = 1;
    27→    27→
    28→    28→/**
    29→    29→ * Build an IEnumMember-shaped value object for ItemDropdown / AutoDropdown.
    30→    30→ * displayName is a plain string (DisplayText form of DisplayNameGetter).
    31→    31→ */
    32→    32→function enumItem(value: string, displayName: string): powerbi.IEnumMember {
    33→    33→    return { value, displayName } as powerbi.IEnumMember;
    34→    34→}
    35→    35→
    36→    36→/**
    37→    37→ * Read a string out of an ItemDropdown's `value` (IEnumMember). Falls back to
    38→    38→ * the provided default when the slice isn't populated yet.
    39→    39→ */
    40→    40→function readEnumValue(
    41→    41→    slice: formattingSettings.ItemDropdown,
    42→    42→    defaultValue: string
    43→    43→): string {
    44→    44→    const v = slice && slice.value;
    45→    45→    if (v && typeof v === "object" && typeof (v as any).value === "string") {
    46→    46→        return (v as any).value as string;
    47→    47→    }
    48→    48→    if (typeof v === "string") {
    49→    49→        return v;
    50→    50→    }
    51→    51→    return defaultValue;
    52→    52→}
    53→    53→
    54→    54→/**
    55→    55→ * Read a boolean slice value (ToggleSwitch). Falls back to default when the
    56→    56→ * slice isn't populated yet (Power BI passes undefined for "default" states).
    57→    57→ */
    58→    58→function readBool(
    59→    59→    slice: formattingSettings.ToggleSwitch | undefined,
    60→    60→    defaultValue: boolean
    61→    61→): boolean {
    62→    62→    if (slice && typeof slice.value === "boolean") {
    63→    63→        return slice.value;
    64→    64→    }
    65→    65→    return defaultValue;
    66→    66→}
    67→    67→
    68→    68→function readNumber(
    69→    69→    slice: formattingSettings.NumUpDown | undefined,
    70→    70→    defaultValue: number
    71→    71→): number {
    72→    72→    if (slice && typeof slice.value === "number" && isFinite(slice.value)) {
    73→    73→        return slice.value;
    74→    74→    }
    75→    75→    return defaultValue;
    76→    76→}
    77→    77→
    78→    78→function readString(
    79→    79→    slice:
    80→    80→        | formattingSettings.TextInput
    81→    81→        | formattingSettings.FontPicker
    82→    82→        | undefined,
    83→    83→    defaultValue: string
    84→    84→): string {
    85→    85→    if (slice && typeof slice.value === "string" && slice.value.length > 0) {
    86→    86→        return slice.value;
    87→    87→    }
    88→    88→    return defaultValue;
    89→    89→}
    90→    90→
    91→    91→function readColor(
    92→    92→    slice: formattingSettings.ColorPicker | undefined,
    93→    93→    defaultValue: string
    94→    94→): string {
    95→    95→    if (slice && slice.value && typeof slice.value.value === "string") {
    96→    96→        return slice.value.value;
    97→    97→    }
    98→    98→    return defaultValue;
    99→    99→}

  1030→  1030→    const gradCount = readNumber(model.bars.gradientColorCount, 2);
  1031→  1031→    const gradColors = [
  1032→  1032→        readColor(model.bars.gradientColor1, "#1F2A5A"),
  1033→  1033→        readColor(model.bars.gradientColor2, "#5B6BA8"),
  1034→  1034→        readColor(model.bars.gradientColor3, "#AFBDE1"),

  1036→  1036→        readColor(model.bars.gradientColor5, "#F0F3FA"),
  1037→  1037→    ].slice(0, Math.max(2, Math.min(5, gradCount)));
  1038→  1038→
  1039→  1039→    return {
  1040→  1040→        bars: {
  1041→  1041→            show: readBool(model.bars.show, true),
  1042→  1042→            barGap: readNumber(model.bars.barGap, 20),
