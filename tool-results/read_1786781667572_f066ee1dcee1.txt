    21→/**
    22→ * powerbi.visuals.ValidatorType is a `const enum` (Min=0, Max=1, Required=2).
    23→ * The build chain inlines it, which would break under isolatedModules if we
    24→ * referenced the enum symbol. We use numeric literals instead.
    25→ */
    26→const VT_MIN = 0;
    27→const VT_MAX = 1;
    28→
    29→/**
    30→ * Build an IEnumMember-shaped value object for ItemDropdown / AutoDropdown.
    31→ * displayName is a plain string (DisplayText form of DisplayNameGetter).
    32→ */
    33→function enumItem(value: string, displayName: string): powerbi.IEnumMember {
    34→    return { value, displayName } as powerbi.IEnumMember;
    35→}
    36→
    37→/**
    38→ * Read a string out of an ItemDropdown's `value` (IEnumMember). Falls back to
    39→ * the provided default when the slice isn't populated yet.
    40→ */
    41→function readEnumValue(
    42→    slice: formattingSettings.ItemDropdown,
    43→    defaultValue: string
    44→): string {
    45→    const v = slice && slice.value;
    46→    if (v && typeof v === "object" && typeof (v as any).value === "string") {
    47→        return (v as any).value as string;
    48→    }
    49→    if (typeof v === "string") {
    50→        return v;
    51→    }
    52→    return defaultValue;
    53→}
    54→
    55→/**
    56→ * Read a boolean slice value (ToggleSwitch). Falls back to default when the
    57→ * slice isn't populated yet (Power BI passes undefined for "default" states).
    58→ */
    59→function readBool(
    60→    slice: formattingSettings.ToggleSwitch | undefined,
    61→    defaultValue: boolean
    62→): boolean {
    63→    if (slice && typeof slice.value === "boolean") {
    64→        return slice.value;
    65→    }
    66→    return defaultValue;
    67→}
    68→
    69→function readNumber(
    70→    slice: formattingSettings.NumUpDown | undefined,
    71→    defaultValue: number
    72→): number {
    73→    if (slice && typeof slice.value === "number" && isFinite(slice.value)) {
    74→        return slice.value;
    75→    }
    76→    return defaultValue;
    77→}
    78→
    79→function readString(
    80→    slice:
    81→        | formattingSettings.TextInput
    82→        | formattingSettings.FontPicker
    83→        | undefined,
    84→    defaultValue: string
    85→): string {
    86→    if (slice && typeof slice.value === "string" && slice.value.length > 0) {
    87→        return slice.value;
    88→    }
    89→    return defaultValue;
    90→}
    91→
    92→function readColor(
    93→    slice: formattingSettings.ColorPicker | undefined,
    94→    defaultValue: string
    95→): string {
    96→    if (slice && slice.value && typeof slice.value.value === "string") {
    97→        return slice.value.value;
    98→    }
    99→    return defaultValue;
   100→}
