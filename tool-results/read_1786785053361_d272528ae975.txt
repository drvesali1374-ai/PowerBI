    20→/**
    21→ * powerbi.visuals.ValidatorType is a `const enum` (Min=0, Max=1, Required=2).
    22→ * The build chain inlines it, which would break under isolatedModules if we
    23→ * referenced the enum symbol. We use numeric literals instead.
    24→ */
    25→const VT_MIN = 0;
    26→const VT_MAX = 1;
    27→
    28→/**
    29→ * Build an IEnumMember-shaped value object for ItemDropdown / AutoDropdown.
    30→ * displayName is a plain string (DisplayText form of DisplayNameGetter).
    31→ */
    32→function enumItem(value: string, displayName: string): powerbi.IEnumMember {
    33→    return { value, displayName } as powerbi.IEnumMember;
    34→}
    35→
    36→/**
    37→ * Read a string out of an ItemDropdown's `value` (IEnumMember). Falls back to
    38→ * the provided default when the slice isn't populated yet.
    39→ */
    40→function readEnumValue(
    41→    slice: formattingSettings.ItemDropdown,
    42→    defaultValue: string
    43→): string {
    44→    const v = slice && slice.value;
    45→    if (v && typeof v === "object" && typeof (v as any).value === "string") {
    46→        return (v as any).value as string;
    47→    }
    48→    if (typeof v === "string") {
    49→        return v;
    50→    }
    51→    return defaultValue;
    52→}
    53→
    54→/**
    55→ * Read a boolean slice value (ToggleSwitch). Falls back to default when the
    56→ * slice isn't populated yet (Power BI passes undefined for "default" states).
    57→ */
    58→function readBool(
    59→    slice: formattingSettings.ToggleSwitch | undefined,
    60→    defaultValue: boolean
    61→): boolean {
    62→    if (slice && typeof slice.value === "boolean") {
    63→        return slice.value;
    64→    }
    65→    return defaultValue;
    66→}
    67→
    68→function readNumber(
    69→    slice: formattingSettings.NumUpDown | undefined,
    70→    defaultValue: number
    71→): number {
    72→    if (slice && typeof slice.value === "number" && isFinite(slice.value)) {
    73→        return slice.value;
    74→    }
    75→    return defaultValue;
    76→}
    77→
    78→function readString(
    79→    slice:
    80→        | formattingSettings.TextInput
    81→        | formattingSettings.FontPicker
    82→        | undefined,
    83→    defaultValue: string
    84→): string {
    85→    if (slice && typeof slice.value === "string" && slice.value.length > 0) {
    86→        return slice.value;
    87→    }
    88→    return defaultValue;
    89→}
    90→
    91→function readColor(
    92→    slice: formattingSettings.ColorPicker | undefined,
    93→    defaultValue: string
    94→): string {
    95→    if (slice && slice.value && typeof slice.value.value === "string") {
    96→        return slice.value.value;
    97→    }
    98→    return defaultValue;
    99→}
