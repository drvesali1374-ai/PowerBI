    18→/** Monotonic counter for unique clip-path IDs (NOT Math.random — predictable). */
    19→let clipIdCounter = 0;
    20→function nextClipId(prefix: string): string {
    21→    clipIdCounter += 1;
    22→    return `${prefix}-${clipIdCounter}`;
    23→}

    25→/** Shared canvas for measureText — created lazily, reused. */
    26→let measureCanvas: HTMLCanvasElement | null = null;
    27→function getMeasureCanvas(): HTMLCanvasElement {
    28→    if (!measureCanvas) {
    29→        measureCanvas = document.createElement("canvas");
    30→        const ctx = measureCanvas.getContext("2d");
    31→        if (ctx) {
    32→            ctx.font = "11px 'Segoe UI', sans-serif";
    33→        }
    34→    }
    35→    return measureCanvas;
    36→}

    38→/** Measure pixel width of a text string with given font. */
    39→function measureTextWidth(
    40→    text: string,
    41→    fontFamily: string,
    42→    fontSize: number
    43→): number {
    44→    const canvas = getMeasureCanvas();
    45→    const ctx = canvas.getContext("2d");
    46→    if (!ctx) return text.length * fontSize * 0.55;
    47→    ctx.font = `${fontSize}px '${fontFamily}', sans-serif`;
    48→    return ctx.measureText(text).width;
    49→}

    51→/** Convert "#RRGGBB" to its rgb triplet for interpolation. */
    52→function hexToRgb(hex: string): [number, number, number] {
    53→    let h = hex.replace("#", "").trim();
    54→    if (h.length === 3) {
    55→        h = h
    56→            .split("")
    57→            .map((c) => c + c)
    58→            .join("");
    59→    }
    60→    const n = parseInt(h, 16);
    61→    if (isNaN(n)) return [0, 0, 0];
    62→    return [(n >> 16) & 0xff, (n >> 8) & 0xff, n & 0xff];
    63→}
    64→
    65→function rgbToHex(r: number, g: number, b: number): string {
    66→    const h = (v: number) =>
    67→        Math.max(0, Math.min(255, Math.round(v)))
    68→            .toString(16)
    69→            .padStart(2, "0");
    70→    return `#${h(r)}${h(g)}${h(b)}`;
    71→}

    73→/**
    74→ * Multi-stop gradient interpolation: given N colors, return the interpolated
    75→ * color at position `t` (0..1). Used to color bars by Pareto rank.
    76→ */
    77→function interpolateGradient(
    78→    colors: string[],
    79→    t: number
    80→): string {
    81→    if (colors.length === 0) return "#1F77B4";
    82→    if (colors.length === 1) return colors[0];
    83→    const clamped = Math.max(0, Math.min(1, t));
    84→    const segments = colors.length - 1;
    85→    const scaled = clamped * segments;
    86→    const idx = Math.min(segments - 1, Math.floor(scaled));
    87→    const localT = scaled - idx;
    88→    const c1 = hexToRgb(colors[idx]);
    89→    const c2 = hexToRgb(colors[idx + 1]);
    90→    return rgbToHex(
    91→        c1[0] + (c2[0] - c1[0]) * localT,
    92→        c1[1] + (c2[1] - c1[1]) * localT,
    93→        c1[2] + (c2[2] - c1[2]) * localT
    94→    );
    95→}

    97→/**
    98→ * Resolve the fill color of a bar.
    99→ * - Selected: settings.bars.selectedColor (v3.0.0.0 behavior — selected bar
   100→ *   gets the dedicated "selected color").
   101→ * - Default mode: settings.bars.defaultColor
   102→ * - Single mode: settings.bars.singleColor
   103→ * - Gradient mode: interpolated across categories by rank
   104→ * - Category mode: leaves it up to the caller (we fall back to default)
   105→ */
   106→function resolveBarColor(
   107→    point: ParetoDataPoint,
   108→    settings: ResolvedSettings,
   109→    totalPoints: number
   110→): string {
   111→    if (point.isSelected) {
   112→        return settings.bars.selectedColor;
   113→    }
   114→    const mode = settings.bars.colorMode;
   115→    switch (mode) {
   116→        case "Single":
   117→            return settings.bars.singleColor;
   118→        case "Gradient": {
   119→            const colors = settings.bars.gradientColors;
   120→            if (colors.length === 0) return settings.bars.defaultColor;
   121→            // Position by rank: t=0 for the first bar, t=1 for the last.
   122→            const t =
   123→                totalPoints > 1 ? (point.rank - 1) / (totalPoints - 1) : 0;
   124→            return interpolateGradient(colors, t);
   125→        }
   126→        case "Category":
   127→            // Category-based coloring isn't persisted across renders here;
   128→            // we fall back to the default color (consistent v3.0 behavior).
   129→            return settings.bars.defaultColor;
   130→        case "Default":
   131→        default:
   132→            return settings.bars.defaultColor;
   133→    }
   134→}
   135→
   136→/**
   137→ * Resolve the opacity of a bar.
   138→ * - Selected: 1.0 (full opacity — v3.0.0.0 behavior).
   139→ * - Dimmed (unselected in a selection state): settings.bars.dimmedOpacity / 100.
   140→ * - Otherwise: settings.bars.opacity / 100.
   141→ */
   142→function resolveBarOpacity(
   143→    point: ParetoDataPoint,
   144→    settings: ResolvedSettings
   145→): number {
   146→    if (point.isSelected) return 1;
   147→    if (point.isDimmed) return settings.bars.dimmedOpacity / 100;
   148→    return settings.bars.opacity / 100;
   149→}

   151→/** Build the dash-array for a given line style + width. */
   152→function dashArray(style: "Solid" | "Dashed" | "Dotted", width: number): string {
   153→    switch (style) {
   154→        case "Dashed":
   155→            return `${width * 3},${width * 2}`;
   156→        case "Dotted":
   157→            return `${width},${width}`;
   158→        case "Solid":
   159→        default:
   160→            return "none";
   161→    }
   162→}

   166→    return formatPercentage(value, decimals);

   198→/** Build a class string for a bar based on its selection state. */
   199→function barClass(point: ParetoDataPoint): string {
   200→    const parts = ["bar"];
   201→    if (point.isSelected) parts.push("selected");
   202→    else if (point.isDimmed) parts.push("dimmed");
   203→    return parts.join(" ");
   204→}

   206→/**
   207→ * Main entry point. Renders the full Pareto chart into the provided SVG.
   208→ * Wipes all existing children first.
   209→ */
   210→export function renderPareto(opts: RenderOptions): void {
   211→    const { svg, model, settings, viewport, isRtl } = opts;
   212→    while (svg.firstChild) svg.removeChild(svg.firstChild);
   213→
   214→    const svgSel = d3.select(svg) as SVGSel;
   215→    svgSel
   216→        .attr("class", "industrial-pareto")
   217→        .attr("width", viewport.width)
   218→        .attr("height", viewport.height)
   219→        .attr("viewBox", `0 0 ${viewport.width} ${viewport.height}`)
   220→        .attr("dir", isRtl ? "rtl" : "ltr");
   221→
   222→    // Background
   223→    if (settings.background.show) {
   224→        const alpha = 1 - settings.background.transparency / 100;
   225→        svgSel
   226→            .append("rect")
   227→            .attr("x", 0)
   228→            .attr("y", 0)
   229→            .attr("width", viewport.width)
   230→            .attr("height", viewport.height)
   231→            .attr("fill", settings.background.color)
   232→            .attr("fill-opacity", alpha);
   233→    }
   234→
   235→    if (!model.hasData || model.points.length === 0) {
   236→        svgSel
   237→            .append("text")
   238→            .attr("x", viewport.width / 2)
   239→            .attr("y", viewport.height / 2)
   240→            .attr("text-anchor", "middle")
   241→            .attr("fill", "#888")
   242→            .style("font-family", "Segoe UI, sans-serif")
   243→            .style("font-size", "14px")
   244→            .text("No data to display");
   245→        return;
   246→    }
   247→
   248→    const points = model.points;
   249→    const totalPoints = points.length;
   250→
   251→    /* ------------------------- Layout / padding ------------------------- */
   252→    // Default paddings. We then EXPAND them based on axis visibility and
   253→    // rotated X labels.
   254→    const padTop = 16;
   255→    const padRight = settings.secondaryAxis.show ? 56 : 24;
   256→    const padLeftBase = settings.yAxis.show ? 60 : 16;
   257→
   258→    // Measure Y-axis tick widths to size the left padding correctly.
   259→    const yMax = model.maxValue;
   260→    const yTickValues = computeNiceTicks(0, yMax, 5);
   261→    let yTickWidth = 0;
   262→    if (settings.yAxis.show) {
   263→        for (const v of yTickValues) {
   264→            const lbl = formatAxisValue(
   265→                v,
   266→                settings.yAxis.valueFormat,
   267→                model.looksLikeDuration
   268→            );
   269→            const w = measureTextWidth(
   270→                lbl,
   271→                settings.yAxis.fontFamily,
   272→                settings.yAxis.fontSize
   273→            );
   274→            if (w > yTickWidth) yTickWidth = w;
   275→        }
   276→    }
   277→
   278→    const padLeft = padLeftBase + yTickWidth;
   279→
   280→    // X-axis label height calculation depends on rotation.
   281→    const innerWidth = Math.max(
   282→        50,
   283→        viewport.width - padLeft - padRight
   284→    );
   285→
   286→    let xLabelHeight = settings.xAxis.fontSize + 8;
   287→    if (settings.xAxis.show) {
   288→        const rotation = settings.xAxis.textRotation;
   289→        const fontFamily = settings.xAxis.fontFamily;
   290→        const fontSize = settings.xAxis.fontSize;
   291→        if (rotation === "Rotation0") {
   292→            xLabelHeight = fontSize + 8;
   293→        } else if (rotation === "Rotation45") {
   294→            // Estimate height from longest label length (rotated 45°).
   295→            let longest = 0;
   296→            for (const p of points) {
   297→                const w = measureTextWidth(
   298→                    p.category,
   299→                    fontFamily,
   300→                    fontSize
   301→                );
   302→                if (w > longest) longest = w;
   303→            }
   304→            xLabelHeight = longest * 0.707 + 8;
   305→        } else {
   306→            // Rotation90 — full height of the longest label.
   307→            let longest = 0;
   308→            for (const p of points) {
   309→                const w = measureTextWidth(
   310→                    p.category,
   311→                    fontFamily,
   312→                    fontSize
   313→                );
   314→                if (w > longest) longest = w;
   315→            }
   316→            xLabelHeight = longest + 8;
   317→        }
   318→    }
   319→
   320→    const padBottom = (settings.xAxis.show ? xLabelHeight : 16) + 12;
   321→    const innerHeight = Math.max(
   322→        40,
   323→        viewport.height - padTop - padBottom
   324→    );
   325→
   326→    /* ----------------------------- Clip paths --------------------------- */
   327→    const plotClipId = nextClipId("plotClip");
   328→    const defs = svgSel.append("defs");
   329→    defs.append("clipPath")
   330→        .attr("id", plotClipId)
   331→        .append("rect")
   332→        .attr("x", 0)
   333→        .attr("y", 0)
   334→        .attr("width", innerWidth)
   335→        .attr("height", innerHeight);
   336→
   337→    /* --------------------------- Root groups ---------------------------- */
   338→    // Plot group is clipped (so bars/curve don't spill into axes).
   339→    // Axes group is NOT clipped (so labels can extend slightly).
   340→    const plotGroup = svgSel
   341→        .append("g")
   342→        .attr(
   343→            "transform",
   344→            `translate(${padLeft},${padTop})`
   345→        )
   346→        .attr("clip-path", `url(#${plotClipId})`);
   347→
   348→    const axesGroup = svgSel
   349→        .append("g")
   350→        .attr("transform", `translate(${padLeft},${padTop})`);
   351→
   352→    /* ------------------------------ Scales ------------------------------ */
   353→    // Bars on X use ScaleBand with configurable gap.
   354→    const gapFraction = Math.max(0, Math.min(0.9, settings.bars.barGap / 100));
   355→    const xScale = d3
   356→        .scaleBand<string>()
   357→        .domain(points.map((p) => p.category))
   358→        .range(isRtl ? [innerWidth, 0] : [0, innerWidth])
   359→        .padding(gapFraction);
   360→
   361→    // Y axis: linear, value domain.
   362→    const yScale = d3
   363→        .scaleLinear<number, number>()
   364→        .domain([0, yMax * 1.05 || 1])
   365→        .range([innerHeight, 0]);
   366→
   367→    // Y2 axis: cumulative percentage (0..100, or user-configured min/max).
   368→    const y2Min = settings.secondaryAxis.min;
   369→    const y2Max = settings.secondaryAxis.max;
   370→    const y2Scale = d3
   371→        .scaleLinear<number, number>()
   372→        .domain([y2Min, y2Max])
   373→        .range([innerHeight, 0]);
   374→
   375→    /* ------------------------------ Grid lines -------------------------- */
   376→    const gridGroup = plotGroup.append("g").attr("class", "grid");
   377→    gridGroup
   378→        .selectAll("line.grid-y")
   379→        .data(yTickValues)
   380→        .enter()
   381→        .append("line")
   382→        .attr("class", "grid-y")
   383→        .attr("x1", 0)
   384→        .attr("x2", innerWidth)
   385→        .attr("y1", (d) => yScale(d))
   386→        .attr("y2", (d) => yScale(d));
   387→
   388→    /* ------------------------------ Bars -------------------------------- */
   389→    const barsGroup = plotGroup.append("g").attr("class", "bars");
   390→    const barSel = barsGroup
   391→        .selectAll<SVGRectElement, ParetoDataPoint>("rect.bar")
   392→        .data(points)
   393→        .enter()
   394→        .append("rect")
   395→        .attr("class", (d) => barClass(d))
   396→        .attr(
   397→            "x",
   398→            (d) => (xScale(d.category) as number) + xScale.bandwidth() * 0
   399→        )
   400→        .attr("y", (d) => yScale(Math.max(0, d.rawValue)))
   401→        .attr("width", xScale.bandwidth())
   402→        .attr("height", (d) => {
   403→            const y = yScale(Math.max(0, d.rawValue));
   404→            return Math.max(0, innerHeight - y);
   405→        })
   406→        .attr("fill", (d) => resolveBarColor(d, settings, totalPoints))
   407→        .attr("fill-opacity", (d) => resolveBarOpacity(d, settings))
   408→        .style("cursor", "pointer");
   409→
   410→    // Wire bar events. The visual owns the handlers so it can dispatch
   411→    // to its selection manager / tooltip service.
   412→    barSel.each(function (point) {
   413→        const barEl = this as SVGRectElement;
   414→        barEl.addEventListener("click", (ev) => {
   415→            opts.onBarClick(point, ev as MouseEvent, barEl);
   416→        });
   417→        barEl.addEventListener("mousemove", (ev) => {
   418→            opts.onBarMouseMove(point, ev as MouseEvent, barEl);
   419→        });
   420→        barEl.addEventListener("mouseleave", () => {
   421→            opts.onBarMouseLeave();
   422→        });
   423→    });
   424→
   425→    /* -------------------------- Bar data labels ------------------------- */
   426→    if (settings.barDataLabels.show) {
   427→        const dl = settings.barDataLabels;
   428→        const labelGroup = plotGroup
   429→            .append("g")
   430→            .attr("class", "bar-labels")
   431→            .style("pointer-events", "none");
   432→        const labelSel = labelGroup
   433→            .selectAll<SVGTextElement, ParetoDataPoint>("text.bar-label")
   434→            .data(points)
   435→            .enter()
   436→            .append("text")
   437→            .attr("class", "bar-label")
   438→            .attr("font-family", dl.fontFamily)
   439→            .attr("font-size", dl.fontSize)
   440→            .attr("fill", dl.color);
   441→
   442→        labelSel.each(function (point) {
   443→            const txt = this as SVGTextElement;
   444→            const v = point.rawValue;
   445→            const text = formatAxisValue(v, dl.format, model.looksLikeDuration);
   446→            const bandX = xScale(point.category) as number;
   447→            const bandW = xScale.bandwidth();
   448→            const y0 = yScale(Math.max(0, v));
   449→            let labelX: number;
   450→            let labelY: number;
   451→            let anchor: "start" | "middle" | "end";
   452→            switch (dl.position) {
   453→                case "Above":
   454→                    labelX = bandX + bandW / 2;
   455→                    labelY = y0 - 6;
   456→                    anchor = "middle";
   457→                    break;
   458→                case "Below":
   459→                    labelX = bandX + bandW / 2;
   460→                    labelY = y0 + dl.fontSize + 2;
   461→                    anchor = "middle";
   462→                    break;
   463→                case "InsideEnd":
   464→                    labelX = bandX + bandW / 2;
   465→                    labelY = y0 + dl.fontSize;
   466→                    anchor = "middle";
   467→                    break;
   468→                case "InsideBase":
   469→                    labelX = bandX + bandW / 2;
   470→                    labelY = innerHeight - 4;
   471→                    anchor = "middle";
   472→                    break;
   473→                default:
   474→                    labelX = bandX + bandW / 2;
   475→                    labelY = y0 - 6;
   476→                    anchor = "middle";
   477→            }
   478→            // Background rect (if enabled). Rendered as a sibling rect BEFORE
   479→            // the text element so it sits underneath.
   480→            if (dl.showBackground) {
   481→                const w = measureTextWidth(
   482→                    text,
   483→                    dl.fontFamily,
   484→                    dl.fontSize
   485→                );
   486→                const padX = 4;
   487→                const padY = 1;
   488→                const bgAlpha = 1 - dl.backgroundTransparency / 100;
   489→                const bgRect = d3
   490→                    .select(txt.parentNode as SVGGElement)
   491→                    .insert("rect", "text.bar-label")
   492→                    .attr("x", labelX - w / 2 - padX)
   493→                    .attr("y", labelY - dl.fontSize - padY)
   494→                    .attr("width", w + padX * 2)
   495→                    .attr("height", dl.fontSize + padY * 2)
   496→                    .attr("fill", dl.backgroundColor)
   497→                    .attr("fill-opacity", bgAlpha)
   498→                    .style("pointer-events", "none");
   499→                void bgRect; // satisfy unused-warning linters
   500→            }
   501→            txt.setAttribute("x", String(labelX));
   502→            txt.setAttribute("y", String(labelY));
   503→            txt.setAttribute("text-anchor", anchor);
   504→            if (containsRtl(point.category)) {
   505→                txt.setAttribute("direction", "rtl");
   506→                txt.setAttribute("unicode-bidi", "plaintext");
   507→                txt.classList.add("rtl-text");
   508→            }
   509→            txt.textContent = text;
   510→        });
   511→    }
   512→
   513→    /* ---------------------------- Cumulative line ----------------------- */
   514→    if (settings.cumulativeLine.show) {
   515→        const cl = settings.cumulativeLine;
   516→        // N+1 vertices: starting point at (0, 0%) plus one vertex at the right
   517→        // edge of each bar.
   518→        const linePoints: Array<[number, number]> = [];
   519→        linePoints.push([0, y2Scale(0)]);
   520→        for (const p of points) {
   521→            const bandX = xScale(p.category) as number;
   522→            const bandW = xScale.bandwidth();
   523→            const xRight = bandX + bandW;
   524→            const y = y2Scale(p.cumulativePercentage);
   525→            linePoints.push([xRight, y]);
   526→        }
   527→
   528→        const linePath = d3
   529→            .line<[number, number]>()
   530→            .x((d) => d[0])
   531→            .y((d) => d[1]);
   532→
   533→        const curveGroup = plotGroup.append("g").attr("class", "cumulative");
   534→        const pathSel = curveGroup
   535→            .append("path")
   536→            .attr("class", "cumulative-line")
   537→            .attr("d", linePath(linePoints) || "")
   538→            .attr("fill", "none")
   539→            .attr("stroke", cl.color)
   540→            .attr("stroke-width", cl.width)
   541→            .attr("stroke-dasharray", dashArray(cl.lineStyle, cl.width));
   542→
   543→        void pathSel; // (kept for future styling hooks)
   544→
   545→        // Markers at right edge of each bar
   546→        if (cl.showMarkers) {
   547→            const markerGroup = curveGroup
   548→                .append("g")
   549→                .attr("class", "cumulative-markers");
   550→            for (let i = 0; i < points.length; i++) {
   551→                const p = points[i];
   552→                const [x, y] = linePoints[i + 1];
   553→                const m = markerGroup
   554→                    .append("g")
   555→                    .attr("class", "cumulative-marker")
   556→                    .attr("transform", `translate(${x},${y})`);
   557→                drawMarker(
   558→                    m,
   559→                    cl.markerShape,
   560→                    cl.markerSize,
   561→                    cl.markerColor
   562→                );
   563→            }
   564→        }
   565→
   566→        // Percentage labels (v3.0.0.0: fixed 1 decimal place, font size +
   567→        // color from settings.cumulativeLine).
   568→        if (cl.showLabels) {
   569→            const labelGroup = curveGroup
   570→                .append("g")
   571→                .attr("class", "cumulative-labels")
   572→                .style("pointer-events", "none");
   573→            for (let i = 0; i < points.length; i++) {
   574→                const p = points[i];
   575→                const [x, y] = linePoints[i + 1];
   576→                const labelText = formatPercent(
   577→                    p.cumulativePercentage,
   578→                    1
   579→                );
   580→                const text = labelGroup
   581→                    .append("text")
   582→                    .attr("class", "cumulative-label")
   583→                    .attr("x", x)
   584→                    .attr("y", y - cl.markerSize - 4)
   585→                    .attr("text-anchor", "middle")
   586→                    .attr("font-size", cl.labelFontSize)
   587→                    .attr("fill", cl.labelColor)
   588→                    .text(labelText);
   589→                if (cl.labelShowBackground) {
   590→                    const w = measureTextWidth(
   591→                        labelText,
   592→                        "Segoe UI",
   593→                        cl.labelFontSize
   594→                    );
   595→                    const padX = 4;
   596→                    const padY = 1;
   597→                    const bgAlpha = 1 - cl.labelBackgroundTransparency / 100;
   598→                    labelGroup
   599→                        .insert("rect", "text.cumulative-label")
   600→                        .attr("x", x - w / 2 - padX)
   601→                        .attr("y", y - cl.markerSize - 4 - cl.labelFontSize - padY)
   602→                        .attr("width", w + padX * 2)
   603→                        .attr("height", cl.labelFontSize + padY * 2)
   604→                        .attr("fill", cl.labelBackgroundColor)
   605→                        .attr("fill-opacity", bgAlpha)
   606→                        .style("pointer-events", "none");
   607→                }
   608→                void text;
   609→            }
   610→        }
   611→    }
   612→
   613→    /* ---------------------------- Reference line ------------------------ */
   614→    if (settings.referenceLine.show) {
   615→        const rl = settings.referenceLine;
   616→        const refY = y2Scale(rl.value);
   617→        const refGroup = plotGroup.append("g").attr("class", "reference");
   618→        if (refY >= 0 && refY <= innerHeight) {
   619→            refGroup
   620→                .append("line")
   621→                .attr("class", "reference-line")
   622→                .attr("x1", 0)
   623→                .attr("x2", innerWidth)
   624→                .attr("y1", refY)
   625→                .attr("y2", refY)
   626→                .attr("stroke", rl.color)
   627→                .attr("stroke-width", rl.width)
   628→                .attr("stroke-dasharray", dashArray(rl.dashStyle, rl.width));
   629→        }
   630→        // Reference label anchored to RIGHT of plot, independent of Y2 axis.
   631→        refGroup
   632→            .append("text")
   633→            .attr("class", "reference-label")
   634→            .attr("x", innerWidth - 4)
   635→            .attr("y", Math.max(11, refY - 4))
   636→            .attr("text-anchor", "end")
   637→            .attr("fill", rl.color)
   638→            .text(formatPercent(rl.value, 0));
   639→    }
   640→
   641→    /* -------------------------------- X axis ---------------------------- */
   642→    if (settings.xAxis.show) {
   643→        const xa = settings.xAxis;
   644→        const xAxisGroup = axesGroup
   645→            .append("g")
   646→            .attr("class", "axis x-axis")
   647→            .attr(
   648→                "transform",
   649→                `translate(0,${innerHeight})`
   650→            );
   651→        // Axis line
   652→        if (xa.showLine) {
   653→            xAxisGroup
   654→                .append("line")
   655→                .attr("x1", 0)
   656→                .attr("x2", innerWidth)
   657→                .attr("y1", 0)
   658→                .attr("y2", 0)
   659→                .attr("stroke", xa.lineColor)
   660→                .attr("stroke-width", xa.lineWidth);
   661→        }
   662→
   663→        // Tick labels: thin according to labelDensity.
   664→        const rotation = xa.textRotation;
   665→        const density = xa.labelDensity;
   666→        const step = computeLabelStep(totalPoints, density);
   667→
   668→        const labels = xAxisGroup
   669→            .selectAll<SVGTextElement, ParetoDataPoint>("text.x-label")
   670→            .data(points)
   671→            .enter()
   672→            .append("text")
   673→            .attr("class", "x-label")
   674→            .attr("font-family", xa.fontFamily)
   675→            .attr("font-size", xa.fontSize)
   676→            .attr("fill", xa.fontColor);
   677→
   678→        labels.each(function (point, idx) {
   679→            const txt = this as SVGTextElement;
   680→            const bandX = xScale(point.category) as number;
   681→            const bandW = xScale.bandwidth();
   682→            const cx = bandX + bandW / 2;
   683→
   684→            // Thin according to labelDensity (only for Rotation0).
   685→            const skip =
   686→                rotation === "Rotation0" && idx % step !== 0;
   687→            if (skip) {
   688→                (this as SVGTextElement).style.display = "none";
   689→                return;
   690→            }
   691→            txt.setAttribute("x", String(cx));
   692→            txt.setAttribute("y", String(xa.fontSize + 4));
   693→            txt.setAttribute("text-anchor", rotation === "Rotation0" ? "middle" : "end");
   694→            if (rotation === "Rotation45") {
   695→                txt.setAttribute(
   696→                    "transform",
   697→                    `rotate(-45 ${cx} ${xa.fontSize + 4})`
   698→                );
   699→            } else if (rotation === "Rotation90") {
   700→                txt.setAttribute(
   701→                    "transform",
   702→                    `rotate(-90 ${cx} ${xa.fontSize + 4})`
   703→                );
   704→            }
   705→            if (containsRtl(point.category)) {
   706→                txt.setAttribute("direction", "rtl");
   707→                txt.setAttribute("unicode-bidi", "plaintext");
   708→                txt.classList.add("rtl-text");
   709→            }
   710→            txt.textContent = point.category;
   711→        });
   712→    }
   713→
   714→    /* -------------------------------- Y axis ---------------------------- */
   715→    if (settings.yAxis.show) {
   716→        const ya = settings.yAxis;
   717→        const yAxisGroup = axesGroup
   718→            .append("g")
   719→            .attr("class", "axis y-axis");
   720→
   721→        if (ya.showLine) {
   722→            yAxisGroup
   723→                .append("line")
   724→                .attr("x1", 0)
   725→                .attr("x2", 0)
   726→                .attr("y1", 0)
   727→                .attr("y2", innerHeight)
   728→                .attr("stroke", ya.lineColor)
   729→                .attr("stroke-width", ya.lineWidth);
   730→        }
   731→
   732→        // Tick labels
   733→        const yLabels = yAxisGroup
   734→            .selectAll<SVGTextElement, number>("text.y-label")
   735→            .data(yTickValues)
   736→            .enter()
   737→            .append("text")
   738→            .attr("class", "y-label")
   739→            .attr("x", -8)
   740→            .attr("y", (d) => yScale(d))
   741→            .attr("dy", "0.32em")
   742→            .attr("text-anchor", "end")
   743→            .attr("font-family", ya.fontFamily)
   744→            .attr("font-size", ya.fontSize)
   745→            .attr("fill", ya.fontColor)
   746→            .text((d) =>
   747→                formatAxisValue(d, ya.valueFormat, model.looksLikeDuration)
   748→            );
   749→
   750→        void yLabels;
   751→
   752→        // Axis title (rotated -90)
   753→        if (ya.title && ya.title.length > 0) {
   754→            axesGroup
   755→                .append("text")
   756→                .attr("class", "axis-title y-axis-title")
   757→                .attr(
   758→                    "transform",
   759→                    `translate(${-yTickWidth - 28},${innerHeight / 2}) rotate(-90)`
   760→                )
   761→                .attr("text-anchor", "middle")
   762→                .attr("font-family", ya.fontFamily)
   763→                .attr("font-size", ya.fontSize + 1)
   764→                .attr("fill", ya.fontColor)
   765→                .text(ya.title);
   766→        }
   767→    }
   768→
   769→    /* --------------------------- Secondary axis ------------------------- */
   770→    if (settings.secondaryAxis.show) {
   771→        const sa = settings.secondaryAxis;
   772→        const y2AxisGroup = axesGroup
   773→            .append("g")
   774→            .attr("class", "axis y2-axis")
   775→            .attr("transform", `translate(${innerWidth},0)`);
   776→
   777→        if (sa.showLine) {
   778→            y2AxisGroup
   779→                .append("line")
   780→                .attr("x1", 0)
   781→                .attr("x2", 0)
   782→                .attr("y1", 0)
   783→                .attr("y2", innerHeight)
   784→                .attr("stroke", sa.lineColor)
   785→                .attr("stroke-width", sa.lineWidth);
   786→        }
   787→
   788→        const y2Ticks = computeNiceTicks(sa.min, sa.max, 5);
   789→        const y2Labels = y2AxisGroup
   790→            .selectAll<SVGTextElement, number>("text.y2-label")
   791→            .data(y2Ticks)
   792→            .enter()
   793→            .append("text")
   794→            .attr("class", "y2-label")
   795→            .attr("x", 8)
   796→            .attr("y", (d) => y2Scale(d))
   797→            .attr("dy", "0.32em")
   798→            .attr("text-anchor", "start")
   799→            .attr("font-family", sa.fontFamily)
   800→            .attr("font-size", sa.fontSize)
   801→            .attr("fill", sa.fontColor)
   802→            .text((d) => `${d.toFixed(0)}%`);
   803→
   804→        void y2Labels;
   805→    }
   806→
   807→    /* --------- Background click handler on SVG (clear selection) -------- */
   808→    svg.addEventListener("click", (ev) => {
   809→        const target = ev.target as Element;
   810→        if (target && target.tagName.toLowerCase() === "svg") {
   811→            opts.onBackgroundClick(ev as MouseEvent);
   812→        }
   813→    });
   814→}

   816→/** Draw a marker (circle/square/triangle/diamond) inside a <g>. */
   817→function drawMarker(
   818→    parent: GroupSel,
   819→    shape: string,
   820→    size: number,
   821→    color: string
   822→): void {
   823→    const r = size / 2;
   824→    switch (shape) {
   825→        case "Square":
   826→            parent
   827→                .append("rect")
   828→                .attr("x", -r)
   829→                .attr("y", -r)
   830→                .attr("width", size)
   831→                .attr("height", size)
   832→                .attr("fill", color);
   833→            break;
   834→        case "Triangle":
   835→            parent
   836→                .append("polygon")
   837→                .attr(
   838→                    "points",
   839→                    `0,${-r} ${r},${r} ${-r},${r}`
   840→                )
   841→                .attr("fill", color);
   842→            break;
   843→        case "Diamond":
   844→            parent
   845→                .append("polygon")
   846→                .attr(
   847→                    "points",
   848→                    `0,${-r} ${r},0 0,${r} ${-r},0`
   849→                )
   850→                .attr("fill", color);
   851→            break;
   852→        case "Circle":
   853→        default:
   854→            parent
   855→                .append("circle")
   856→                .attr("r", r)
   857→                .attr("fill", color);
   858→    }
   859→}

   861→/** Compute "nice" tick values for a [min, max] range. Pure numeric, no D3 axis. */
   862→function computeNiceTicks(min: number, max: number, count: number): number[] {
   863→    if (min === max) {
   864→        return [min];
   865→    }
   866→    const span = max - min;
   867→    const step0 = span / count;
   868→    const mag = Math.pow(10, Math.floor(Math.log10(step0)));
   869→    const norm = step0 / mag;
   870→    let step: number;
   871→    if (norm < 1.5) step = 1 * mag;
   872→    else if (norm < 3) step = 2 * mag;
   873→    else if (norm < 7) step = 5 * mag;
   874→    else step = 10 * mag;
   875→    const start = Math.ceil(min / step) * step;
   876→    const ticks: number[] = [];
   877→    for (let v = start; v <= max + step * 0.5; v += step) {
   878→        ticks.push(Math.round(v * 1e9) / 1e9);
   879→    }
   880→    return ticks;
   881→}

   883→/** Compute the visible-label step for labelDensity thinning. */
   884→function computeLabelStep(
   885→    total: number,
   886→    density: "ShowAll" | "Auto" | "Sparse"
   887→): number {
   888→    if (density === "ShowAll") return 1;
   889→    if (total <= 8) return 1;
   890→    if (density === "Auto") {
   891→        // ~10 labels max
   892→        return Math.max(1, Math.ceil(total / 10));
   893→    }
   894→    // Sparse: ~5 labels max
   895→    return Math.max(1, Math.ceil(total / 5));
   896→}

   900→/**
   901→ * Lightweight selection-state sync. Updates isSelected/isDimmed on the model
   902→ * and patches the CSS class + fill + fill-opacity on each existing bar.
   903→ *
   904→ * v3.0.0.0 behavior:
   905→ * - Selected bar: fill = settings.bars.selectedColor, fill-opacity = 1
   906→ * - Dimmed bar: fill-opacity = settings.bars.dimmedOpacity / 100
   907→ *   (fill color stays the bar's natural color, just dimmed)
   908→ */
   909→export function syncSelectionState(
   910→    svg: SVGSVGElement,
   911→    model: ParetoViewModel,
   912→    settings: ResolvedSettings
   913→): void {
   914→    const totalPoints = model.points.length;
   915→    const sel = d3.select(svg);
   916→    sel.selectAll<SVGRectElement, ParetoDataPoint>("rect.bar").each(function (
   917→        point
   918→    ) {
   919→        const bar = d3.select(this);
   920→        bar.attr("class", barClass(point));
   921→        bar.attr("fill", resolveBarColor(point, settings, totalPoints));
   922→        bar.attr("fill-opacity", resolveBarOpacity(point, settings));
   923→    });
   924→}
